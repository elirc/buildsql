// Merges authored expansion volumes (content-src/<volume>/*.json) into the
// published track packs in public/content, assigning ids and orders and
// refreshing index.json counts.
import { readdir, readFile, writeFile } from 'node:fs/promises'
import { resolve } from 'node:path'

const root = resolve(import.meta.dirname, '..')
const contentDir = resolve(root, 'public', 'content')
const volume = process.argv[2]
if (!volume) {
  console.error('usage: node scripts/merge-content.mjs <volume-dir-name>')
  process.exit(1)
}
const srcDir = resolve(root, 'content-src', volume)
const files = (await readdir(srcDir)).filter((name) => name.endsWith('.json')).sort()
if (!files.length) {
  console.error(`no authored files in ${srcDir}`)
  process.exit(1)
}

const index = JSON.parse(await readFile(resolve(contentDir, 'index.json'), 'utf8'))
const packs = new Map()
for (const meta of index.tracks) {
  packs.set(meta.id, JSON.parse(await readFile(resolve(contentDir, `${meta.id}.json`), 'utf8')))
}

const problems = []
const skipped = []
let added = 0
let addedMissions = 0

const text = (value) => typeof value === 'string' && value.trim().length > 0

function incompleteReason(lesson) {
  for (const field of ['title', 'summary', 'briefing', 'explanation', 'fullReadme', 'fullReading']) {
    if (!text(lesson[field])) return `missing ${field}`
  }
  const missions = lesson.exercises ?? []
  if (!missions.some((mission) => mission.kind === 'core')) return 'no core missions'
  for (const mission of missions) {
    if (!text(mission.id) || !text(mission.prompt) || !text(mission.hints) || !text(mission.importantHints)) {
      return `mission ${mission.id ?? '?'} missing text`
    }
    if (![100, 140, 250].includes(mission.xp)) return `mission ${mission.id} bad xp`
    const references = Object.values(mission.references ?? {})
    if (!references.length || !references.every(text)) return `mission ${mission.id} missing reference`
  }
  return null
}

for (const file of files) {
  const data = JSON.parse(await readFile(resolve(srcDir, file), 'utf8'))
  const pack = packs.get(data.track)
  if (!pack) {
    problems.push(`${file}: unknown track ${data.track}`)
    continue
  }
  const seen = new Set(pack.lessons.map((lesson) => lesson.slug))
  for (const lesson of data.lessons) {
    if (!lesson.slug || seen.has(lesson.slug)) {
      problems.push(`${file}: duplicate or missing slug ${lesson.slug}`)
      continue
    }
    const incomplete = incompleteReason(lesson)
    if (incomplete) {
      skipped.push(`${file}:${lesson.slug} - ${incomplete}`)
      continue
    }
    seen.add(lesson.slug)
    const order = pack.lessons.length
    pack.lessons.push({
      id: `${data.track}/${lesson.slug}`,
      slug: lesson.slug,
      order,
      title: lesson.title,
      summary: lesson.summary,
      briefing: lesson.briefing,
      explanation: lesson.explanation,
      fullReadme: lesson.fullReadme,
      fullReading: lesson.fullReading,
      exercises: lesson.exercises,
    })
    added += 1
    addedMissions += lesson.exercises.length
  }
}

if (skipped.length) {
  console.warn(`Skipped ${skipped.length} incomplete sector(s):`)
  for (const entry of skipped.slice(0, 40)) console.warn(`- ${entry}`)
}

if (problems.length) {
  console.error(`merge aborted with ${problems.length} problem(s):`)
  for (const problem of problems.slice(0, 30)) console.error(`- ${problem}`)
  process.exit(1)
}

let lessonTotal = 0
let missionTotal = 0
for (const meta of index.tracks) {
  const pack = packs.get(meta.id)
  meta.lessonCount = pack.lessons.length
  meta.exerciseCount = pack.lessons.reduce((sum, lesson) => sum + lesson.exercises.length, 0)
  meta.coreCount = pack.lessons.reduce(
    (sum, lesson) => sum + lesson.exercises.filter((mission) => mission.kind === 'core').length,
    0,
  )
  lessonTotal += meta.lessonCount
  missionTotal += meta.exerciseCount
  await writeFile(resolve(contentDir, `${meta.id}.json`), JSON.stringify(pack))
}
index.lessonCount = lessonTotal
index.exerciseCount = missionTotal
await writeFile(resolve(contentDir, 'index.json'), `${JSON.stringify(index, null, 2)}\n`)

console.log(`Merged ${volume}: +${added} sectors, +${addedMissions} missions -> ${lessonTotal} sectors, ${missionTotal} missions.`)
